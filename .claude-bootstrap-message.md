## Patrón de push (Python via mcp__workspace__bash)

```python
cd /sessions/<session>/mnt/OMM-ERP
python3 <<'EOF'
import base64, json, urllib.request
PAT = open('.claude-pat').read().strip()
REPO = "EliasMicha/omm-erp"
FILE = "src/pages/X.tsx"
with open(FILE, "rb") as f: b64 = base64.b64encode(f.read()).decode()
def api(m, p, payload=None):
    url = f"https://api.github.com{p}"
    h = {"Authorization": f"Bearer {PAT}", "Accept": "application/vnd.github+json"}
    d = json.dumps(payload).encode() if payload else None
    return json.loads(urllib.request.urlopen(urllib.request.Request(url, data=d, headers=h, method=m)).read())
blob = api("POST", f"/repos/{REPO}/git/blobs", {"content": b64, "encoding": "base64"})
ref = api("GET", f"/repos/{REPO}/git/ref/heads/main")
latest = ref["object"]["sha"]
ci = api("GET", f"/repos/{REPO}/git/commits/{latest}")
tree = api("POST", f"/repos/{REPO}/git/trees", {"base_tree": ci["tree"]["sha"], "tree": [{"path": FILE, "mode": "100644", "type": "blob", "sha": blob["sha"]}]})
nc = api("POST", f"/repos/{REPO}/git/commits", {"message": "...", "tree": tree["sha"], "parents": [latest]})
api("PATCH", f"/repos/{REPO}/git/refs/heads/main", {"sha": nc["sha"]})
print(f"Commit: {nc['sha'][:7]}")
EOF
```

Para múltiples archivos en un commit: agregar más entries al array `tree`.
